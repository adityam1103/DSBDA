Practical 13: Weather Average Reducer Snippet   Javapublic void reduce(Text key, Iterable<Text> values, Context context) throws IOException, InterruptedException {
    int tempSum = 0, count = 0;
    for (Text val : values) {
        String[] nums = val.toString().split(",");
        tempSum += Integer.parseInt(nums[0]);
        count++;
    }
    double avgTemp = (double) tempSum / count;
    context.write(key, new Text("AvgTemp=" + avgTemp));
}
